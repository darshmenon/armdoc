cmake_minimum_required(VERSION 3.8)
project(armdoc_cpp)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(TinyXML2 REQUIRED)

add_executable(armdoc_fast src/armdoc_fast_node.cpp)
target_include_directories(armdoc_fast PRIVATE include)
ament_target_dependencies(armdoc_fast rclcpp sensor_msgs std_msgs controller_manager_msgs)
target_link_libraries(armdoc_fast tinyxml2::tinyxml2)

add_executable(armdoc_core_cli src/core_cli.cpp)
target_include_directories(armdoc_core_cli PRIVATE include)

install(TARGETS armdoc_fast armdoc_core_cli DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_package()
